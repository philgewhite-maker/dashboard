(async () => {
  // A messages page shows this profile next to a sidebar list of every
  // other match's small avatar — verified against a real captured page,
  // everything belonging to THIS profile sits inside <main id="main-content">,
  // with the sidebar in a separate <aside>. Scoping every lookup to that
  // root keeps someone else's photo from ending up in this person's record.
  const root = document.getElementById('main-content') || document;
  const $$ = (sel, r) => [...(r || root).querySelectorAll(sel)];
  const GROUP_HEADERS = new Set(['Essentials', 'Lifestyle', 'Interests', 'Messages', 'Photos', 'More about me']);

  // Some sections (a photo carousel's other slides, apparently also fields
  // further down a long profile — verified: a second captured page had
  // "Lives in London" and "160 cm" but nothing for Languages, which the
  // profile visibly had) only render into the DOM once scrolled into view.
  // Scrolling the whole panel first, before reading anything, is the same
  // fix already needed for photos, just applied earlier.
  const scrollBox = [...document.querySelectorAll('div')]
    .filter((d) => d.scrollHeight > d.clientHeight + 300 && d.clientHeight > 200)
    .sort((a, b) => b.scrollHeight - a.scrollHeight)[0];
  if (scrollBox) {
    for (let y = 0; y < scrollBox.scrollHeight; y += 400) {
      scrollBox.scrollTop = y;
      await new Promise((r) => setTimeout(r, 200));
    }
    scrollBox.scrollTop = 0;
    await new Promise((r) => setTimeout(r, 200));
  }

  // Verified against a real captured profile page: name and age share one
  // <h1> whose aria-label is "{name} {age} years".
  const h1 = root.querySelector('h1[aria-label]');
  let name = '', age = '';
  if (h1) {
    const m = (h1.getAttribute('aria-label') || '').match(/^(.+?)\s+(\d+)\s+years?$/);
    if (m) { name = m[1].trim(); age = m[2]; } else { name = h1.textContent.trim(); }
  }

  const fields = [];
  // Set once the chat-history block below finds it, so the photo scan
  // further down can exclude it — a sender's own small avatar next to
  // each chat message is a real gotinder.com background-image too, just
  // not one of THIS profile's photos (confirmed live: the user's own
  // avatar from their sent messages ended up saved as one of a match's
  // photos).

  // "You matched with X on DD/MM/YYYY" — unverified against real markup
  // (no captured page has had an open chat yet), so this deliberately
  // searches by TEXT CONTENT rather than guessing a structural selector.
  // Safe either way: if the wording differs, this just finds nothing,
  // same as any other field this snippet doesn't recognise.
  const matchedEl = [...root.querySelectorAll('*')].find((el) => el.children.length === 0 && /you matched.*on\s+\d{1,2}\/\d{1,2}\/\d{2,4}/i.test(el.textContent || ''));
  if (matchedEl) {
    const dm = matchedEl.textContent.match(/on\s+(\d{1,2})\/(\d{1,2})\/(\d{2,4})/i);
    // Tinder shows DD/MM/YYYY; the connection field is an <input type="date">,
    // which needs ISO (YYYY-MM-DD) to actually populate rather than sit blank.
    if (dm) fields.push({ label: 'Matched on', value: `${dm[3]}-${dm[2].padStart(2, '0')}-${dm[1].padStart(2, '0')}` });
  }
  const seen = new Set();
  // Every "vital" (kids, education, smoking, pets, communication style,
  // zodiac, languages, and whatever else Tinder adds) is an icon named
  // .../descriptors/{type}@2x.png inside an <li> with its own <h3>
  // label — generic by design, so it picks up whatever this profile
  // filled in without hardcoding each field.
  $$('img[src*="/descriptors/"]').forEach((img) => {
    const li = img.closest('li');
    if (!li) return;
    const h3 = li.querySelector('h3');
    const label = h3 ? h3.textContent.trim() : '';
    const valueEl = [...li.querySelectorAll('div')].find((d) => d.textContent.trim() && !d.querySelector('img') && !d.querySelector('h3'));
    const value = valueEl ? valueEl.textContent.trim() : '';
    if (label && value && !seen.has(label)) { fields.push({ label, value }); seen.add(label); }
  });

  // A second, unlabeled strip right under the name (verified against a
  // second captured page) — height, job+employer, school, city, pronouns —
  // each an inline <svg> icon (not a descriptors/ png) followed by ONE
  // value div with no <h3> of its own, so the label has to come from the
  // shape of the text itself rather than a heading next to it.
  // Distance is bucketed rather than kept as raw "2 miles away" text — the
  // dashboard sorts connections by these bucket labels, and a bucket is what
  // makes hundreds of matches rankable by proximity at a glance instead of
  // comparing exact numbers. Both units are baked into the label (some
  // profiles show miles, some km, depending on the person's own settings,
  // not anything this snippet controls) so the bucket reads the same either
  // way. <= not <: an app only ever shows whole-number distances, so a
  // strict "< 2" would never actually match a real "2 miles away" and that
  // bucket would sit permanently empty. Keep this list's boundaries in sync
  // with DISTANCE_BUCKET_ORDER in connections.js by hand — a console
  // snippet can't import from it.
  const DISTANCE_BUCKETS = [2, 5, 10, 20, 30, 50, 100, 1000, 2000];
  function bucketDistance(mi) {
    for (const b of DISTANCE_BUCKETS) if (mi <= b) return `≤${b}mi/${Math.round(b * 1.60934)}km`;
    const roundedMi = Math.round(mi);
    return `${roundedMi}mi/${Math.round(mi * 1.60934)}km+`;
  }

  // Height likewise shows both units regardless of which one Tinder
  // rendered for this profile, rather than being unreadable/uncomparable
  // whenever two profiles happen to use different units.
  function formatHeight(text) {
    const cmMatch = text.match(/^(\d+)\s*cm$/);
    const ftMatch = text.match(/^(\d+)'(\d+)"?$/);
    let cm = null;
    if (cmMatch) cm = parseInt(cmMatch[1], 10);
    else if (ftMatch) cm = Math.round((parseInt(ftMatch[1], 10) * 12 + parseInt(ftMatch[2], 10)) * 2.54);
    if (cm === null) return text;
    const totalIn = cm / 2.54;
    const ft = Math.floor(totalIn / 12);
    const inch = Math.round(totalIn % 12);
    return `${cm}cm / ${ft}'${inch}"`;
  }

  // enumerated list is actually reliable here (unlike names/interests,
  // which are open-ended). Verified live: it renders in the SAME unlabeled
  // basics strip as height/job/city, not the Essentials icon+<h3> pattern
  // — so without this it fell into the generic "Profile detail" catch-all
  // indistinguishably from the person's gender identity, school name, etc.
  const ORIENTATIONS = new Set(['Straight', 'Gay', 'Lesbian', 'Bisexual', 'Pansexual', 'Asexual', 'Queer', 'Demisexual']);
  // Same reasoning as ORIENTATIONS: gender identity is also a fixed picker
  // in Tinder's UI, rendered in this same unlabeled strip with no text
  // label of its own — without an enumerated list it's indistinguishable
  // from a stray "Profile detail".
  const GENDERS = new Set(['Man', 'Woman', 'Non-binary', 'Trans man', 'Trans woman', 'Transgender', 'Genderqueer', 'Genderfluid', 'Agender', 'Bigender', 'Pangender', 'Demiboy', 'Demigirl', 'Two-Spirit', 'Intersex']);
  // Languages usually render as a comma-joined list (caught below by the
  // comma check), but a person who only speaks one language shows a bare
  // single name with nothing to structurally distinguish it from a job
  // title or hobby — so a single language needs its own name list too.
  // Not exhaustive; anything missing just falls to the generic catch-all
  // below rather than being mis-labelled.
  const LANGUAGES = new Set(['English', 'French', 'Spanish', 'German', 'Italian', 'Portuguese', 'Mandarin', 'Cantonese', 'Chinese', 'Japanese', 'Korean', 'Arabic', 'Russian', 'Polish', 'Dutch', 'Turkish', 'Hindi', 'Urdu', 'Bengali', 'Punjabi', 'Gujarati', 'Marathi', 'Tamil', 'Telugu', 'Kannada', 'Malayalam', 'Vietnamese', 'Thai', 'Indonesian', 'Malay', 'Tagalog', 'Filipino', 'Greek', 'Hebrew', 'Swedish', 'Norwegian', 'Danish', 'Finnish', 'Czech', 'Hungarian', 'Romanian', 'Bulgarian', 'Ukrainian', 'Croatian', 'Serbian', 'Slovak', 'Slovenian', 'Lithuanian', 'Latvian', 'Estonian', 'Icelandic', 'Irish', 'Welsh', 'Catalan', 'Basque', 'Swahili', 'Amharic', 'Yoruba', 'Zulu', 'Afrikaans', 'Persian', 'Farsi', 'Pashto', 'Nepali', 'Sinhala', 'Burmese', 'Khmer', 'Lao', 'Mongolian', 'Georgian', 'Armenian', 'Azerbaijani', 'Kazakh', 'Uzbek', 'Albanian', 'Bosnian', 'Macedonian', 'Luxembourgish', 'Maltese', 'Esperanto', 'Latin']);

  const seenText = new Set();
  root.querySelectorAll('li').forEach((li) => {
    if (li.querySelector('img[src*="/descriptors/"]') || li.querySelector('h3') || !li.querySelector('svg')) return;
    const valueEl = [...li.querySelectorAll('div')].find((d) => d.textContent.trim() && d.children.length === 0);
    if (!valueEl) return;
    const text = valueEl.textContent.trim();
    if (!text || seenText.has(text)) return;
    seenText.add(text);
    const livesIn = text.match(/^Lives in (.+)$/);
    const away = text.match(/^([\d.]+)\s*(miles?|km)\s*away$/i);
    if (/^\d+\s*cm$/.test(text) || /^\d+'\d+"?$/.test(text)) fields.push({ label: 'Height', value: formatHeight(text) });
    else if (livesIn) fields.push({ label: 'City', value: livesIn[1] });
    else if (away) {
      const n = parseFloat(away[1]);
      const miles = /km/i.test(away[2]) ? n * 0.621371 : n;
      fields.push({ label: 'Distance', value: bucketDistance(miles) });
    }
    else if (/^(She|He|They)\/(Her|Him|Them)$/i.test(text)) fields.push({ label: 'Pronouns', value: text });
    else if (/ at /.test(text)) fields.push({ label: 'Job', value: text });
    // A basics-strip item with a comma is, structurally, the only
    // multi-value one Tinder shows this way — verified live, this is how
    // Languages actually renders (NOT the Essentials icon+<h3> pattern
    // this snippet originally assumed it would share with Zodiac/Education).
    else if (text.includes(',')) fields.push({ label: 'Languages', value: text });
    else if (LANGUAGES.has(text)) fields.push({ label: 'Languages', value: text });
    else if (ORIENTATIONS.has(text)) fields.push({ label: 'Orientation', value: text });
    else if (GENDERS.has(text)) fields.push({ label: 'Gender', value: text });
    else fields.push({ label: 'Profile detail', value: text });
  });

  // Interest chips: a <ul aria-labelledby="X"> where h2#X is "Interests"
  // (verified against a second captured page). A chip Tinder highlights as
  // something you both like uses a distinct brand-colour background rather
  // than the plain neutral one every other chip uses — that's the only
  // signal separating "Travel (shared)" from an ordinary "Theatre".
  const interestsHeading = $$('h2').find((h) => h.textContent.trim() === 'Interests');
  if (interestsHeading && interestsHeading.id) {
    const list = root.querySelector(`ul[aria-labelledby="${interestsHeading.id}"]`);
    if (list) {
      const chips = [...list.querySelectorAll('li')].map((li) => {
        const div = li.querySelector('div');
        const text = (li.querySelector('span') || li).textContent.trim();
        if (!text) return '';
        const shared = div && !(div.className || '').includes('background-secondary');
        return shared ? `${text} (shared)` : text;
      }).filter(Boolean);
      if (chips.length) fields.push({ label: 'Interests', value: chips.join(', ') });
    }
  }

  // Chat history: a "Conversation history with {name}" heading whose value
  // block is the ENTIRE thread's flattened textContent — verified
  // against a real captured page, where without this it swept every
  // message, phone numbers and composer UI chrome into one field via the
  // generic h2 pass below. Each message carries a visually-hidden "Sent at
  // HH:MM" accessibility label immediately followed by the same time again,
  // the sender, and a colon — a reliable, repeating anchor confirmed
  // across every message in that real transcript. A "DD/MM/YYYY, HH:MM"
  // divider (and sometimes a bare "Sent" delivery-status word right before
  // it, no space) sits between some messages and isn't part of either
  // one's text. Parsed messages are shown in the review screen before
  // anything saves, never trusted silently — misattributing who said
  // what is worse than a merely misfiled field.
  const chatHeading = $$('h2').find((h) => /^Conversation history with /i.test(h.textContent.trim()));
  if (chatHeading) {
    const card = chatHeading.closest('section') || (chatHeading.parentElement && chatHeading.parentElement.parentElement);
    chatCard = card;
    const valueBlock = card && [...card.children].find((el) => !el.contains(chatHeading) && el.textContent.trim());
    const raw = valueBlock ? valueBlock.textContent : '';
    const cut = raw.replace(/Reply to .+/s, '');
    const msgRe = /Sent at (\d{1,2}:\d{2})\d{1,2}:\d{2}([^:]{1,40}):/g;
    const matches = [...cut.matchAll(msgRe)];
    const messages = [];
    for (let i = 0; i < matches.length; i++) {
      const m = matches[i];
      const start = m.index + m[0].length;
      const end = i + 1 < matches.length ? matches[i + 1].index : cut.length;
      const body = cut.slice(start, end)
        .replace(/(Sent)?\d{1,2}\/\d{1,2}\/\d{4},?\s*\d{1,2}:\d{2}\s*$/, '')
        .replace(/Sent\s*$/, '')
        .trim();

      if (sender && body) messages.push(`[${m[1]}] ${sender}: ${body}`);
    }
    if (messages.length) fields.push({ label: 'Chat history', value: messages.join('\n') });
  }

  // "Looking for" and prompt answers ("My biography would be called: ...")
  // share a shape too: an <h2> label with its answer nearby. Skips known
  // section headers, which wrap a LIST rather than carrying one value.
  // "Looking for" specifically opens with an emoji in its own span before
  // the actual text (verified: grabbing only the first leaf span there
  // caught nothing but a single heart-eyes emoji) — take the whole value
  // block's text and strip a leading emoji, rather than the first fragment.
  $$('h2').forEach((h2) => {
    const label = h2.textContent.trim();

    if (!card) return;
    const valueBlock = [...card.children].find((el) => !el.contains(h2) && el.textContent.trim());

    value = value.replace(/^\p{Extended_Pictographic}️?\s*/u, '').trim();
    if (value && value !== label) { fields.push({ label, value }); seen.add(label); }
    // "Looking for" carries a second, separate answer below it — a
    // relationship-type chip (e.g. "Monogamy") sitting in its own div
    // after the main value, not inside a <ul> the way Interest chips
    // are (verified against a real captured page).
    if (label === 'Looking for') {
      const chip = [...card.querySelectorAll('div[class*="Bdrs(30px)"]')].find((d) => !d.closest('ul') && d.textContent.trim());
      if (chip) fields.push({ label: 'Relationship type', value: chip.textContent.trim() });
    }
  });

  // Photos load from a public CDN with no login needed (verified) —
  // unlike Google Photos, no byte-capture trick is needed. But the carousel
  // only keeps the CURRENTLY shown photo in the DOM at all — the rest
  // are virtualized until you actually click to them (verified: a captured
  // page mid-scroll only ever had ONE of this profile's photos as a real
  // URL, out of 8). So click through every numbered photo tab, collecting
  // whatever newly appears each time, rather than a single one-shot scan.
  //
  // Scanned across the whole panel (root), same as before this session's
  // fixes — a narrower scope (the smallest element wrapping every
  // "Photo N" tab button) was tried and reverted: confirmed live, it broke
  // photo capture entirely, most likely because the actual photo elements
  // live outside whatever that computed ancestor turned out to be, an
  // assumption about the carousel's DOM shape that was never independently
  // verified. chatCard exclusion (a real, verified container) plus the
  // folder-id filter below are the actual defenses against a stray photo
  // from elsewhere in the panel — both are independent of scan scope, so
  // narrowing the scope was never actually necessary for that.
  const photoButtons = $$('button[aria-label^="Photo "]');

  // The same photo can turn up under more than one URL shape in one scan
  // pass — confirmed live, a resized "640x800_75_<uuid>.webp" AND a
  // full-size "original_<uuid>.jpg" of the identical file, which a plain
  // string dedupe doesn't recognise as the same photo. Keyed on the
  // (folder id, uuid) pair every shape of the URL carries, keeping
  // whichever variant was actually seen first.
  const UUID_RE = /[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/i;
  function photoKey(url) {
    const folder = url.match(/gotinder\.com\/([a-zA-Z0-9]+)\//);
    const uuid = url.match(UUID_RE);
    return folder && uuid ? `${folder[1]}/${uuid[0]}` : url;
  }
  const photoMap = new Map();
  const folderCounts = new Map();
  const scanPhotos = () => {
    $$('*').forEach((el) => {
      if (chatCard && chatCard.contains(el)) return;
      const b = getComputedStyle(el).backgroundImage;
      const m = b && b.match(/url\("([^"]+)"\)/);
      if (!m || !m[1].includes('gotinder.com')) return;
      const url = m[1].replace(/\/\d+x\d+_/, '/640x800_');
      const key = photoKey(url);
      if (photoMap.has(key)) return;
      const folderMatch = url.match(/gotinder\.com\/([a-zA-Z0-9]+)\//);
      const folder = folderMatch ? folderMatch[1] : null;
      photoMap.set(key, { url, folder });
      if (folder) folderCounts.set(folder, (folderCounts.get(folder) || 0) + 1);
    });
  };
  scanPhotos();
  for (const btn of photoButtons) {
    btn.click();
    await new Promise((r) => setTimeout(r, 300));
    scanPhotos();
  }

  // Every real photo of one person shares the same Tinder folder id in its
  // URL, confirmed across every profile captured this session — an
  // outlier folder id showing up means a DIFFERENT profile's photo got
  // swept in somehow. Kept to whichever folder id most of the captured
  // photos actually belong to; anything else is dropped.
  let dominantFolder = null, dominantCount = 0;
  folderCounts.forEach((count, folder) => { if (count > dominantCount) { dominantFolder = folder; dominantCount = count; } });
  const photos = [...photoMap.values()].filter((p) => !dominantFolder || !p.folder || p.folder === dominantFolder).map((p) => p.url);

  // The one stable, permanent key back to this exact Tinder match — the
  // id in the page's own URL (/app/messages/<id>), unrelated to name
  // (which can change) or anything else that could drift. Stored on the
  // connection so a later import can check whether it's still in the
  // current match list at all.
  const matchId = (location.pathname.match(/\/app\/messages\/([a-zA-Z0-9]+)/) || [])[1] || '';
  const out = { name, age, fields, photos, matchId };
  const json = JSON.stringify(out, null, 1);
  let copied = false;
  try { copy(json); copied = true; } catch (e) { /* not in a real DevTools console */ }
  if (!copied) {
    try { await navigator.clipboard.writeText(json); copied = true; } catch (e) { /* no clipboard permission/focus */ }
  }
  console.log((copied ? 'Copied — ' : 'Could not auto-copy, select and copy manually — ')
    + `${name || '(no name found)'}, ${age || '?'}: ${fields.length} field${fields.length === 1 ? '' : 's'}, ${out.photos.length} photo${out.photos.length === 1 ? '' : 's'}.`);
  if (!copied) console.log(json);
})();

// Pasting this runs it once with the default batch size below. To run a
// different size without re-pasting, just call it again right in the
// console afterward: tinderBulkImport(30)
window.tinderBulkImport = async (batchSize) => {
  // Same per-profile extraction as the single-profile snippet above, just
  // wrapped as a function so it can run once per match instead of once per
  // page load. Kept identical to that already-tested logic on purpose —
  // this bulk version is the harder-to-verify piece (unattended, many real
  // profiles in a row), so any drift from the known-good version is exactly
  // the kind of mistake most likely to go unnoticed until it's already run
  // 15 times.
  async function extractProfile(expectedName) {
    const root = document.getElementById('main-content') || document;
    const $$ = (sel, r) => [...(r || root).querySelectorAll(sel)];
    const GROUP_HEADERS = new Set(['Essentials', 'Lifestyle', 'Interests', 'Messages', 'Photos', 'More about me']);

    // If the visible profile changes mid-extraction -- the user clicking
    // elsewhere in the tab, or the SPA re-rendering on its own -- reading
    // on would mix that OTHER profile's photos/fields into this one's
    // result. Confirmed live: a different match's photo ended up saved as
    // part of someone else's extract. Checked at every point that spends
    // real time waiting (the scroll settle, each photo-carousel click),
    // not just once before starting, so an interruption anywhere aborts
    // this profile cleanly instead of silently returning contaminated data
    // -- the caller already skips a profile that throws and retries it
    // next run.
    function assertStillHere() {
      const h1El = root.querySelector('h1[aria-label]');
      const label = h1El ? (h1El.getAttribute('aria-label') || '') : '';
      if (!label.toLowerCase().startsWith(String(expectedName || '').toLowerCase())) {
        throw new Error(`Page moved to a different profile mid-extraction (expected "${expectedName}", now showing "${label}") -- aborting rather than mixing up photos or fields.`);
      }
    }

    const scrollBox = [...document.querySelectorAll('div')]
      .filter((d) => d.scrollHeight > d.clientHeight + 300 && d.clientHeight > 200)
      .sort((a, b) => b.scrollHeight - a.scrollHeight)[0];
    if (scrollBox) {
      for (let y = 0; y < scrollBox.scrollHeight; y += 400) {
        scrollBox.scrollTop = y;
        await new Promise((r) => setTimeout(r, 200));
        assertStillHere();
      }
      scrollBox.scrollTop = 0;
      await new Promise((r) => setTimeout(r, 200));
      assertStillHere();
    }

    const h1 = root.querySelector('h1[aria-label]');
    let name = '', age = '';
    if (h1) {
      const m = (h1.getAttribute('aria-label') || '').match(/^(.+?)\s+(\d+)\s+years?$/);
      if (m) { name = m[1].trim(); age = m[2]; } else { name = h1.textContent.trim(); }
    }

    const fields = [];
    let chatCard = null;
    const matchedEl = [...root.querySelectorAll('*')].find((el) => el.children.length === 0 && /you matched.*on\s+\d{1,2}\/\d{1,2}\/\d{2,4}/i.test(el.textContent || ''));
    if (matchedEl) {

        if (!link) { console.warn(`Couldn't find ${row.name} any more -- the sidebar re-rendered. Stopping here.`); break; }
        link.click();

        if (!loaded) {
          console.warn(`${row.name}'s profile didn't load in time -- skipped, will retry next run.`);
          continue;

          // nothing the dashboard review screen would actually show
          // beyond a name to confirm. Not worth a review slot; marked

            console.log(`${row.name}: nothing to add -- skipped.`);
          } else {
            extracted.matchId = row.id; // already known from the sidebar row — same permanent id the single-profile snippet reads from the URL
            profiles.push(extracted);
          }
          doneVersions[row.id] = SCRIPT_VERSION;
          localStorage.setItem(DONE_KEY, JSON.stringify(doneVersions));
        } catch (err) {
          console.warn(`Extraction failed for ${row.name}: ${err.message} -- skipped.`);
        }
        if (i < todo.length - 1) {
          const delay = MIN_DELAY_MS + Math.random() * (MAX_DELAY_MS - MIN_DELAY_MS);
          console.log(`Waiting ${Math.round(delay / 1000)}s...`);
          await new Promise((r) => setTimeout(r, delay));
        }
      }
    } finally {
      const out = { profiles };
      const json = JSON.stringify(out, null, 1);
      let copied = false;
      try { copy(json); copied = true; } catch (e) { /* not in a real DevTools console */ }
      if (!copied) { try { await navigator.clipboard.writeText(json); copied = true; } catch (e) { /* no clipboard permission/focus */ } }
      // Marked done in localStorage the moment a profile is scraped here,
      // not when the dashboard actually saves it -- the two are different
      // origins with no way to signal back. So the clipboard alone isn't
      // enough: closing the tab, or the clipboard getting overwritten,
      // before pasting would lose this batch for good even though it's
      // already marked as never-offer-again. A real file on disk survives
      // that -- still there to paste in later even if today's paste never
      // happens.
      try {
        const blob = new Blob([json], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `tinder-batch-${new Date().toISOString().replace(/[:.]/g, '-')}.json`;
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(() => URL.revokeObjectURL(url), 10000);
      } catch (e) { /* download blocked -- clipboard/console log are still there */ }
      console.log(`${copied ? 'Copied' : 'Could not auto-copy -- select and copy manually'} -- ${profiles.length} of ${todo.length} attempted profile${todo.length === 1 ? '' : 's'} extracted, and saved to a file in your Downloads folder too. Paste into the dashboard now or later.`);
      if (!copied) console.log(json);
    }
  }
};

  // session, so a bare URL 403s the instant it's embedded there, no matter
  // how fresh. Fetching each cover HERE, while still authenticated, and
  // embedding the actual bytes as a data URI sidesteps that for good. A few
  // at a time so it doesn't hammer Google; anything that fails (CORS, a
  // slow connection, whatever) just keeps its plain URL — never worse than
  // today, only ever better.
  if (onAlbums) {
    const withCover = out.albums.filter(a => a.cover && a.cover.startsWith('http'));
    if (withCover.length) {
      console.log(`Fetching ${withCover.length} cover images while logged in — give this a few seconds…`);
      const queue = [...withCover];
      const embedOne = async (a) => {
        try {
          const res = await fetch(a.cover, { credentials: 'include' });
          if (!res.ok) throw new Error(`HTTP ${res.status}`);
          const blob = await res.blob();
          a.cover = await new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = () => resolve(reader.result);
            reader.onerror = reject;
            reader.readAsDataURL(blob);
          });
        } catch (e) { /* leave the plain URL — no worse than before */ }
      };
      const worker = async () => { while (queue.length) await embedOne(queue.shift()); };
      await Promise.all(Array.from({ length: 5 }, worker));
      const embedded = withCover.filter(a => a.cover.startsWith('data:')).length;
      console.log(`Embedded ${embedded} of ${withCover.length} covers` + (embedded < withCover.length ? ' — the rest kept their link and will show as expired in the dashboard.' : '.'));
    }
  }

  const json = JSON.stringify(out, null, 1);
  // copy() is a DevTools convenience global, injected only into a console
  // opened directly in that panel — it throws ReferenceError anywhere else
  // (a docked/undocked console reattached differently, remote debugging,
  // some extensions' console proxies). Clipboard API needs a secure context
  // AND focus, neither guaranteed right after pasting into DevTools, so this
  // tries three ways and always leaves the JSON visible either way.
  let copied = false;
  try { copy(json); copied = true; } catch (e) { /* not in a real DevTools console */ }
  if (!copied) {
    try { await navigator.clipboard.writeText(json); copied = true; } catch (e) { /* no clipboard permission/focus */ }
  }
  console.log(`${copied ? 'Copied' : 'Could not auto-copy — select the JSON below and copy it manually:'} `
    + `${out.albums.length} albums and ${out.people.length} faces.`
    + (onAlbums ? ' Now run this again on /people to spot people with no album.' : ''));
  if (!copied) console.log(json);
})();

Health data (Health Connect)

A browser page has no way to read Health Connect directly — it's a native Android API with no web equivalent. Instead, a bridge app on your phone (e.g. HC Webhook / "Health Connect Webhook" from Google Play) posts whatever data you grant it to health.php on this same server. This panel is a raw viewer for now, showing exactly what's arrived, rather than a finished chart — the actual display gets built once a real payload's shape is known.
Setup: copy server/health.php.example to server/health.php, set $SECRET to the same value as sync.php, upload next to it. In the bridge app, point its webhook target at that URL — append ?secret=<your secret> to it if the app doesn't let you set a custom header.
