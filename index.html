// Every match/message row in either sidebar list shares the same
// /app/messages/<id> URL scheme, and both lists are always in the DOM
// at once — the inactive tab is just transformed off-screen, not
// removed — verified against a real saved page with both tabs

function collectCandidates() {
  const seen = new Set();
  const rows = [];
  const grab = (ul) => {
    if (!ul) return;
    ul.querySelectorAll('li a[href*="/app/messages/"]').forEach((a) => {
      const m = a.href.match(/\/app\/messages\/([a-zA-Z0-9]+)/);
      if (!m) return;
      const id = m[1];
      if (seen.has(id)) return;
      seen.add(id);
      const name = (a.getAttribute('aria-label') || '').replace(/\.\s*Start chat\.?$/i, '').trim();
      rows.push({ id, name, href: a.href });
    });
  };
  grab(document.querySelector('ul[aria-label="Your recent messages"]'));
  grab(document.querySelector('ul[aria-label="Your matches"]'));
  return rows;
}

// Re-finds a row's link fresh at click time, by match id rather than by
// comparing href strings — confirmed live: `a.href` (the JS
// property, always browser-resolved to an absolute URL) doesn't
// necessarily equal the raw href ATTRIBUTE Tinder's live DOM actually
// carries, so `document.querySelector('a[href="' + resolvedUrl + '"]')`
// failed to find a person visibly on screen. Matching on the id parsed
// out of the live `.href` property on both sides (same parsing
// collectCandidates already does) sidesteps that mismatch entirely.
function findLinkById(id) {
  for (const ul of [document.querySelector('ul[aria-label="Your recent messages"]'), document.querySelector('ul[aria-label="Your matches"]')]) {
    if (!ul) continue;
    const found = [...ul.querySelectorAll('li a[href*="/app/messages/"]')].find((a) => {
      const m = a.href.match(/\/app\/messages\/([a-zA-Z0-9]+)/);
      return m && m[1] === id;
    });
    if (found) return found;
  }
  return null;
}

// Polls the profile panel until it's actually showing the person just
// clicked, rather than a fixed delay — SPA navigation time varies,
// and a fixed wait is either too short sometimes or wastefully long
// every time.
async function waitForProfile(expectedName, timeoutMs) {
  const start = Date.now();
  while (Date.now() - start < timeoutMs) {
    const h1 = document.querySelector('#main-content h1[aria-label]');
    const label = h1 ? (h1.getAttribute('aria-label') || '') : '';
    if (label && label.toLowerCase().startsWith(expectedName.toLowerCase())) return true;
    await new Promise((r) => setTimeout(r, 250));
  }
  return false;
}

const SCRIPT_VERSION = 6; // chat dates now come from each message's own timestamp, not a nearby divider guess
const DONE_KEY = 'dashTinderBulkDone';
const MIN_DELAY_MS = 10000, MAX_DELAY_MS = 60000; // randomised, not a fixed interval — a fixed gap is a bot tell

function loadDoneVersions() {
  let doneVersions = {};
  try {
    const raw = JSON.parse(localStorage.getItem(DONE_KEY) || '{}');
    if (Array.isArray(raw)) {
      // Legacy format from before version-stamping existed: a flat array
      // of ids with no version attached. Every one of these could have
      // been scraped under the exact bug this versioning exists to
      // recover from, so treat them all as version 0 -- older than any
      // real version -- and let them be re-offered.
      raw.forEach((id) => { doneVersions[id] = 0; });
    } else {
      doneVersions = raw || {};
    }
  } catch (e) { doneVersions = {}; }
  return doneVersions;
}
function saveDoneVersions(doneVersions) { localStorage.setItem(DONE_KEY, JSON.stringify(doneVersions)); }
function isDoneWith(doneVersions, id) { return (doneVersions[id] || 0) >= SCRIPT_VERSION; }

// Writes whatever's in `list` to its own downloaded file, optionally also
// the clipboard. `state` ({chunkIndex, totalSaved}) belongs to the
// caller, not a shared counter here -- tinderBulkImport() and
// tinderCatchUp() each track their own run's numbering independently, so
// one doesn't pick up a count left over from the other.
function downloadChunk(list, copyToClipboard, state) {
  if (!list.length) return;
  state.chunkIndex++;
  state.totalSaved += list.length;
  const json = JSON.stringify({ profiles: list }, null, 1);
  let copied = false;
  if (copyToClipboard) {
    try { copy(json); copied = true; } catch (e) { /* not in a real DevTools console */ }
    if (!copied) { try { navigator.clipboard.writeText(json); copied = true; } catch (e) { /* no clipboard permission/focus */ } }
  }
  try {
    const blob = new Blob([json], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `tinder-batch-${new Date().toISOString().replace(/[:.]/g, '-')}-part${state.chunkIndex}.json`;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 10000);
    console.log(`Checkpoint: ${list.length} profile${list.length === 1 ? '' : 's'} saved to part${state.chunkIndex} in Downloads (${state.totalSaved} total so far)${copied ? ' -- also on the clipboard' : ''}.`);
  } catch (e) {
    console.warn(`Checkpoint download blocked for part${state.chunkIndex} -- ${list.length} profiles only exist in this tab's memory right now, keep it open.`);
  }
}

// Squaring the random factor skews toward the low end of the range
// instead of a flat spread -- long, near-60s waits get rarer while the
// occasional real one still happens (still not a fixed gap, still not a
// bot tell), cutting ~24% off the average total wait versus a uniform
// pick across the same 10-60s bounds.
async function randomDelay() {
  const delay = MIN_DELAY_MS + Math.pow(Math.random(), 2) * (MAX_DELAY_MS - MIN_DELAY_MS);
  console.log(`Waiting ${Math.round(delay / 1000)}s...`);
  await new Promise((r) => setTimeout(r, delay));
}

window.tinderBulkImport = async (batchSize) => {
  const BATCH_SIZE = batchSize || 15; // "10-20 or so per day" — override per-call: tinderBulkImport(30), or tinderBulkImport(250) left running overnight
  const CHUNK_SIZE = 25; // checkpoint to its own downloaded file this often -- a long unattended run surviving a sleep/crash/closed-tab partway through loses at most this many, not the whole run, and each file is pasteable into the dashboard as soon as it lands rather than waiting for the end

  const doneVersions = loadDoneVersions();
  const candidates = collectCandidates();
  const todo = candidates.filter((r) => !isDoneWith(doneVersions, r.id)).slice(0, BATCH_SIZE);
  const profiles = [];
  const chunkState = { chunkIndex: 0, totalSaved: 0 };

  // Always logged, whether or not there's anything new to actually run --
  // "how far through 205 am I" shouldn't require sitting through another
  // batch to find out.
  const doneCount = candidates.filter((r) => isDoneWith(doneVersions, r.id)).length;
  console.log(`${candidates.length} visible in the sidebar -- ${doneCount} already done, ${candidates.length - doneCount} remaining.`);

  if (!todo.length) {
    console.log(`Nothing new -- all visible matches already processed. Scroll the list down to reveal more, then run again.`);
    return;
  }
  console.log(`Processing ${todo.length} new match${todo.length === 1 ? '' : 'es'}. This takes a while -- real delays are the point.`);
  // Wrapped in try/finally so a crash partway through still outputs and
  // copies whatever WAS gathered, rather than losing a run's worth of
  // waiting to one bad profile.
  try {
    for (let i = 0; i < todo.length; i++) {
      const row = todo[i];
      console.log(`[${i + 1}/${todo.length}] ${row.name}...`);
      const link = findLinkById(row.id);
      if (!link) { console.warn(`Couldn't find ${row.name} any more -- the sidebar re-rendered. Stopping here.`); break; }
      link.click();
      const loaded = await waitForProfile(row.name, 8000);
      if (!loaded) {
        console.warn(`${row.name}'s profile didn't load in time -- skipped, will retry next run.`);
        continue;
      }
      await new Promise((r) => setTimeout(r, 500));
      try {
        const extracted = await extractProfile(row.name);
        // A bare match with no bio, no essentials, no chat, no photos —
        // nothing the dashboard review screen would actually show
        // beyond a name to confirm. Not worth a review slot; marked
        // done and moved past without adding it to the output at all.
        if (!extracted.fields.length && !extracted.photos.length) {
          console.log(`${row.name}: nothing to add -- skipped.`);
        } else {
          extracted.matchId = row.id; // already known from the sidebar row — same permanent id the single-profile snippet reads from the URL
          profiles.push(extracted);
          if (profiles.length >= CHUNK_SIZE) downloadChunk(profiles.splice(0, profiles.length), false, chunkState);
        }
        doneVersions[row.id] = SCRIPT_VERSION;
        saveDoneVersions(doneVersions);
      } catch (err) {
        console.warn(`Extraction failed for ${row.name}: ${err.message} -- skipped.`);
      }
      if (i < todo.length - 1) await randomDelay();
    }
  } finally {
    // Marked done in localStorage the moment a profile is scraped here,
    // not when the dashboard actually saves it -- the two are different
    // origins with no way to signal back. So the clipboard alone isn't
    // enough: closing the tab, or the clipboard getting overwritten,
    // before pasting would lose this batch for good even though it's
    // already marked as never-offer-again. A real file on disk survives
    // that -- still there to paste in later even if today's paste never
    // happens. Whatever's left in `profiles` here is the tail end that
    // hasn't hit a CHUNK_SIZE checkpoint yet -- everyone else already has
    // their own file, downloaded as the run went.
    downloadChunk(profiles, true, chunkState);
    console.log(`${chunkState.totalSaved} of ${todo.length} attempted profile${todo.length === 1 ? '' : 's'} extracted, across ${chunkState.chunkIndex} file${chunkState.chunkIndex === 1 ? '' : 's'} in your Downloads folder. Paste them into the dashboard now or later -- the file picker there takes several at once.`);
  }
};

// Re-checks people ALREADY captured for new activity since a given date,
// instead of only ever picking up brand-new matches like
// tinderBulkImport() does -- someone marked done there is never
// revisited again even after a new message arrives. Built on an
// assumption about Tinder's own sidebar order, per the request that
// prompted this: the Messages tab is sorted by last-message-date, newest
// first, and the Matches tab by match-date, newest first —
// unverified against a real long list, since no captured page has had
// one to check.
//
// Two genuinely different questions, not one applied twice:
//
// Matches = "is this a profile I've never indexed at all?" A match with
// no chat yet has no reliable date signal on the row itself to compare
// against anything -- presence in doneVersions (seen it before, yes/no)
// IS the only signal, so this sweeps with `floor` pinned to EPOCH
// ("date zero") and never even reads a date field: an already-indexed
// match is skipped WITHOUT opening it at all (nothing new a plain
// re-index would find), stopping once several in a row turn out
// already-known. This only ever catches brand-new matches -- it does
// NOT re-check an old match's profile for edits (bio/photos changed with
// no message sent) or notice an unmatch; see the note below on why.
//
// Messages = "does this conversation (new OR already-known) have
// activity since `since`?" Once two people exchange even one message,
// Tinder moves that row here and it carries both the profile AND the
// chat -- so this is also where a profile edit on an already-known match
// would actually surface, IF they've messaged at all. Every row gets
// reopened here, including already-done ones (that's the whole point),
// reading the real "Last message date" and stopping the first time it's
// older than `since`.
//
// Both run through the one sweep() below, just anchored to a different
// floor (EPOCH vs a real date) -- not two copies of the same logic.
//
// Not done: detecting an UNMATCH (someone vanishes from the Matches
// list entirely) needs a full walk of the current list diffed against
// everyone already known as matched -- worth doing once, as a distinct
// tracked attribute, but not worth the weekly cost of walking the whole
// list just for that, so this sweep keeps its early-stop rather than
// running Matches to completion every time.
//
// Either sweep can also hit a safety cap before running out of rows -- a
// guard against the sort-order assumption turning out wrong for some
// account or view. A loud, bounded stop ("hit the cap") beats silently
// missing everything past a wrongly-ordered entry.
window.tinderCatchUp = async (sinceDate, batchSize) => {
  const since = String(sinceDate || '').trim();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(since)) {
    console.warn('tinderCatchUp needs a since date like "2026-08-10" -- e.g. tinderCatchUp("2026-08-10")');
    return;
  }
  const EPOCH = '0000-01-01'; // sentinel for "no date signal, presence-only" -- see the Matches note above

  const CAP = batchSize || 40; // per sweep -- see the safety-cap note above
  const doneVersions = loadDoneVersions();
  const chunkState = { chunkIndex: 0, totalSaved: 0 };
  const profiles = [];
  const visited = new Set();

  async function processRow(row) {
    const link = findLinkById(row.id);
    if (!link) return { stop: true, reason: `Couldn't find ${row.name} any more -- the sidebar re-rendered.` };
    link.click();
    const loaded = await waitForProfile(row.name, 8000);
    if (!loaded) { console.warn(`${row.name}'s profile didn't load in time -- skipped.`); return { skip: true }; }
    await new Promise((r) => setTimeout(r, 500));
    try {
      return { extracted: await extractProfile(row.name) };
    } catch (err) {
      console.warn(`Extraction failed for ${row.name}: ${err.message} -- skipped.`);
      return { skip: true };
    }
  }

  function rowsFrom(ul) {
    return (ul ? [...ul.querySelectorAll('li a[href*="/app/messages/"]')] : []).map((a) => {
      const m = a.href.match(/\/app\/messages\/([a-zA-Z0-9]+)/);
      if (!m) return null;
      return { id: m[1], name: (a.getAttribute('aria-label') || '').replace(/\.\s*Start chat\.?$/i, '').trim() };
    }).filter(Boolean);
  }

  function keep(row, extracted) {
    if (!extracted.fields.length && !extracted.photos.length) {
      console.log(`${row.name}: nothing to add -- skipped.`);
    } else {
      extracted.matchId = row.id;
      profiles.push(extracted);
      if (profiles.length >= 25) downloadChunk(profiles.splice(0, profiles.length), false, chunkState);
    }
    doneVersions[row.id] = SCRIPT_VERSION;
    saveDoneVersions(doneVersions);
  }

  // One sweep, used for both tabs -- see the comment above the function
  // for what `floor` (a real date, or EPOCH) actually changes.
  async function sweep(label, ul, dateFieldLabel, floor) {
    const rows = rowsFrom(ul);
    console.log(`${label}: ${rows.length} visible, scanning from the top for activity since ${floor === EPOCH ? 'the beginning' : floor}...`);
    let n = 0;
    let consecutiveDone = 0;
    for (const row of rows) {
      if (visited.has(row.id)) continue;
      if (floor === EPOCH && isDoneWith(doneVersions, row.id)) {
        consecutiveDone++;
        if (consecutiveDone >= 5) { console.log(`${label}: 5 already-known in a row -- stopping (list is date-sorted, so the rest are presumably older too).`); break; }
        continue;
      }
      consecutiveDone = 0;
      if (n >= CAP) { console.log(`${label}: hit the safety cap of ${CAP} -- stopping here.`); break; }
      n++;
      visited.add(row.id);
      console.log(`${label} [${n}] ${row.name}...`);
      const result = await processRow(row);
      if (result.stop) { console.warn(result.reason); break; }
      if (!result.skip) {
        if (floor === EPOCH) {
          keep(row, result.extracted); // brand-new row -- no date gate needed, just keep it
        } else {
          const dateField = result.extracted.fields.find((f) => f.label === dateFieldLabel);
          if (!dateField) {
            console.log(`${row.name}: no ${dateFieldLabel.toLowerCase()} found -- skipped from this pass.`);
          } else if (dateField.value < floor) {
            console.log(`${row.name}: ${dateFieldLabel.toLowerCase()} ${dateField.value}, before ${floor} -- ${label} is date-sorted, so stopping here.`);
            break;
          } else {
            keep(row, result.extracted);
          }
        }
      }
      await randomDelay();
    }
  }

  try {
    await sweep('Messages', document.querySelector('ul[aria-label="Your recent messages"]'), 'Last message date', since);
    await sweep('Matches', document.querySelector('ul[aria-label="Your matches"]'), null, EPOCH); // dateFieldLabel unused when floor is EPOCH -- see the note above
  } finally {
    downloadChunk(profiles, true, chunkState);
    console.log(`${chunkState.totalSaved} profile${chunkState.totalSaved === 1 ? '' : 's'} with activity since ${since}, across ${chunkState.chunkIndex} file${chunkState.chunkIndex === 1 ? '' : 's'} in your Downloads folder.`);
  }
};

// A one-off, fast check -- no profile opens, no delays between people,
// since it's just comparing two lists of ids already sitting in the DOM
// (once fully scrolled) against everyone this account has ever indexed.
// Unlike tinderBulkImport()/tinderCatchUp(), completeness genuinely
// matters here: collectCandidates() alone only sees whatever's currently
// scrolled into view, so a merely-not-yet-loaded person would look
// identical to a real unmatch. Both sidebars get scrolled to the bottom
// first for exactly that reason -- unverified against a real long list,
// since no captured page has had one to check the scroll behaviour on.
window.tinderFindUnmatched = async () => {
  // Same "find the nearest real scroll container, not necessarily the
  // list element itself" approach used elsewhere in this snippet, just
  // run to actual completion (scrollHeight stops growing) rather than a
  // fixed number of steps, since there's no known list length to aim for
  // up front the way the whole-profile-panel sweep has.
  async function scrollListFully(ul) {
    if (!ul) return;
    let box = ul;
    while (box && !(box.scrollHeight > box.clientHeight + 50 && box.clientHeight > 100)) box = box.parentElement;
    if (!box) return;
    let lastHeight = -1;
    let stable = 0;
    for (let i = 0; i < 200 && stable < 3; i++) {
      box.scrollTop = box.scrollHeight;
      await new Promise((r) => setTimeout(r, 300));
      if (box.scrollHeight === lastHeight) stable++; else stable = 0;
      lastHeight = box.scrollHeight;
    }
  }

  console.log('Scrolling both lists to load everyone first -- this can take a while for a long history...');
  await scrollListFully(document.querySelector('ul[aria-label="Your recent messages"]'));
  await scrollListFully(document.querySelector('ul[aria-label="Your matches"]'));

  const currentIds = new Set(collectCandidates().map((r) => r.id));
  const doneVersions = loadDoneVersions();
  const everKnown = Object.keys(doneVersions);
  const unmatched = everKnown.filter((id) => !currentIds.has(id));

  if (!unmatched.length) {
    console.log(`${currentIds.size} currently visible, ${everKnown.length} ever indexed -- nobody missing.`);
    return;
  }
  const json = JSON.stringify({ unmatched }, null, 1);
  let copied = false;
  try { copy(json); copied = true; } catch (e) { /* not in a real DevTools console */ }
  if (!copied) { try { await navigator.clipboard.writeText(json); copied = true; } catch (e) { /* no clipboard permission/focus */ } }
  console.log(`${unmatched.length} previously-indexed id${unmatched.length === 1 ? '' : 's'} no longer visible in either list -- ${copied ? 'copied' : 'shown below'}. Paste into the dashboard same as any other batch (it only adds a note and a tag, nothing destructive).`);
  if (!copied) console.log(json);
};

console.log('Ready. Run tinderBulkImport() for the default 15, tinderBulkImport(30) for a custom size, or e.g. tinderBulkImport(250) left running overnight -- checkpoints to a file every 25. Or tinderCatchUp("2026-08-10") to re-check active conversations and new matches since a date -- good for a weekly pass. Or tinderFindUnmatched() to spot anyone missing from both lists since they were last indexed.');
})();

(async () => {
  // Accumulates across pages: run on /albums, then on /people, then paste.
  // Bump whenever the capture/scroll logic below changes -- forces
  // window.__dash to start clean instead of quietly carrying forward
  // whatever an OLDER version of this script already captured (including
  // anything it captured wrong, or never caught at all) across repeated
  // runs in the same tab. Same idea as SCRIPT_VERSION in the Tinder
  // import snippet.
  const SNIPPET_VERSION = 4; // /people match is now by aria-label presence, not a base64-alignment-fragile href substring that missed roughly half the tiles
  window.__dash = (window.__dash && window.__dash.v === SNIPPET_VERSION) ? window.__dash : { v: SNIPPET_VERSION, albums: new Map(), people: new Set() };
  const onAlbums = location.pathname.includes('album');
  const grab = () => {
    // Verified against the live page: each album has TWO anchors sharing an
    // href (a cover tile and a text tile), aria-label is absent, and the
    // cover is a CSS background-image on a descendant, not an img. So both
    // anchors are merged by URL, keeping whichever piece each one carries.
    // A SHARED album's tile links to /share/<id>?key=... instead of
    // /album/<id> -- confirmed live from a real Shared-album tile's
    // outerHTML, and the reason every shared album was silently invisible
    // to this snippet, not a scroll/timing miss like the others.
    document.querySelectorAll('a[href*="/album/"], a[href*="/share/"]').forEach(a => {
      // The /share/ link's ?key=... IS the access token -- stripping it
      // (like the plain query-string trim below) would save a URL that
      // 403s the moment it's opened later. /album/ links carry no such
      // token, so they keep the existing trim (whatever transient params
      // Google sometimes appends there aren't part of the link's identity).
      const url = /\/share\//.test(a.href) ? a.href : a.href.split('?')[0];
      const lines = (a.innerText || '').split('\n').map(s => s.trim()).filter(Boolean);
      let cover = '';
      for (const el of a.querySelectorAll('*')) {
        const b = getComputedStyle(el).backgroundImage;
        // Mid-transition this can be a cross-fade of two layers, e.g.
        // -fade(url("A") 8.8%, url("B") ...) — not a plain url("..."), so a
        // fixed-offset slice grabbed the whole thing as garbage. Pull out
        // the first real url(...) instead, wherever it's wrapped.
        const m = b && b.match(/url\("([^"]+)"\)/);
        if (m && m[1].includes('usercontent')) { cover = m[1]; break; }
      }
      // The grid requests tiny thumbnails (often 24px); ask for a usable one.
      if (cover) cover = cover.replace(/=s\d+/, '=s400');
      // A shared album's second line reads "47 items · Shared", not just
      // "47 items" -- the original $-anchored regex only ever matched a
      // private album's plain count line, so a shared album's count (and,
      // worse, its title-exclusion check) silently missed every time.
      const ITEMS_RE = /^\d+\s*items?\b/i;
      const titleLine = lines.find(l => l.toLowerCase() !== 'no items' && l.toLowerCase() !== 'more options' && !ITEMS_RE.test(l));
      const countLine = lines.find(l => ITEMS_RE.test(l));
      const prev = window.__dash.albums.get(url) || {};
      window.__dash.albums.set(url, {
        url,
        title: prev.title || titleLine || '',
        count: prev.count || (countLine ? parseInt(countLine, 10) : 0),
        cover: prev.cover || cover,
      });
    });
    // The old /QUYxUWlw/ check on a.href was a base64 substring match on a
    // length-prefixed protobuf blob -- it only ever matched by coincidence,
    // when the preceding name's byte length happened to leave the token
    // 3-byte-aligned. A shorter or longer name shifts that alignment and
    // produces a DIFFERENT base64 rendering of the exact same underlying
    // token (confirmed against two real tiles: "Phil White" matched,
    // "Tamara White" -- one character shorter -- didn't, same page, same
    // kind of tile). Requiring an aria-label instead is both more reliable
    // and simpler: every real person tile carries one (verified against
    // several real tiles), and it's what a person's name actually lives in
    // regardless of how the href happens to encode.
    document.querySelectorAll('a[href*="/search/"][aria-label]').forEach(a => {
      const t = (a.getAttribute('aria-label') || '').trim();
      if (t && !/^Photo|\d{1,2}:\d{2}/.test(t)) window.__dash.people.add(t);
    });
  };
  const box = [...document.querySelectorAll('div')]
    .filter(d => d.scrollHeight > d.clientHeight + 500 && d.clientHeight > 200)
    .sort((a,b) => b.scrollHeight - a.scrollHeight)[0] || document.scrollingElement;
  const countNow = () => onAlbums ? window.__dash.albums.size : window.__dash.people.size;
  // Two things changed here from a plain fixed-step scroll-and-grab:
  //
  // 1. grab() runs TWICE per stop, a beat apart, not once. A tile's name
  //    label can populate a moment after the tile itself lands in the DOM
  //    (an async render, not a layout one) -- a single grab() right after
  //    scrolling sampled plenty of tiles mid-render with a blank label,
  //    confirmed live on /people as a "select, miss, select, miss" pattern
  //    even though every one of those tiles was clearly a named person on
  //    screen a moment later.
  //
  // 2. On /people specifically (autoStop below), the sweep stops once
  //    several steps in a row add no one new. Named people cluster at the
  //    top with no gaps -- past them is Google's much longer "Others"
  //    section this app has no use for, which used to make the whole
  //    sweep take minutes for nothing. Albums have no such shape (a real
  //    album can be anywhere on the page), so autoStop never applies there.
  const sweep = async (from, to, step, autoStop) => {
    let flat = 0;
    for (let y = from; step > 0 ? y < to : y > to; y += step) {
      const before = countNow();
      grab();
      box.scrollTop = y;
      await new Promise(r => setTimeout(r, 260));
      grab();
      await new Promise(r => setTimeout(r, 220));
      if (autoStop) {
        if (countNow() === before) flat++; else flat = 0;
        if (flat >= 6) break;
      }
    }
    grab();
  };
  box.scrollTop = 0;
  await new Promise(r => setTimeout(r, 400));
  await sweep(0, box.scrollHeight, 120, !onAlbums);
  // Second pass, back upward, over the SAME span the first pass actually
  // covered (not all the way from scrollHeight down to 0) -- same
  // "virtualized out of the grid again before grab() caught it" reasoning
  // that motivated a two-pass sweep in the first place, just bounded to
  // wherever there was actually anything worth finding.
  const reachedY = box.scrollTop;
  await new Promise(r => setTimeout(r, 300));
  await sweep(reachedY, 0, -120, false);
  const out = {
    albums: [...window.__dash.albums.values()].filter(a => a.title),
    people: [...window.__dash.people],
  };
  if (onAlbums && out.albums.length === 0) {
    // Google changes this markup from time to time. Rather than fail
    // silently, show what IS on the page so the selector can be corrected.
    const sample = [...document.querySelectorAll('a')].slice(0, 40)
      .map(a => a.getAttribute('href')).filter(Boolean);
    console.warn('No albums matched. Paste this back to Claude:', JSON.stringify(sample));
  }

  // A cover URL only resolves for whoever is logged in at the moment it's
  // fetched — the dashboard is a different site with no way to borrow this
  // session, so a bare URL 403s the instant it's embedded there, no matter
  // how fresh. Fetching each cover HERE, while still authenticated, and
  // embedding the actual bytes as a data URI sidesteps that for good. A few
  // at a time so it doesn't hammer Google; anything that fails (CORS, a
  // slow connection, whatever) just keeps its plain URL — never worse than
  // today, only ever better.
  if (onAlbums) {
    const withCover = out.albums.filter(a => a.cover && a.cover.startsWith('http'));
    if (withCover.length) {
      console.log(`Fetching ${withCover.length} cover images while logged in — give this a few seconds…`);
      const queue = [...withCover];
      const embedOne = async (a) => {
        try {
          const res = await fetch(a.cover, { credentials: 'include' });
          if (!res.ok) throw new Error(`HTTP ${res.status}`);
          const blob = await res.blob();
          a.cover = await new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = () => resolve(reader.result);
            reader.onerror = reject;
            reader.readAsDataURL(blob);
          });
        } catch (e) { /* leave the plain URL — no worse than before */ }
      };
      const worker = async () => { while (queue.length) await embedOne(queue.shift()); };
      await Promise.all(Array.from({ length: 5 }, worker));
      const embedded = withCover.filter(a => a.cover.startsWith('data:')).length;
      console.log(`Embedded ${embedded} of ${withCover.length} covers` + (embedded < withCover.length ? ' — the rest kept their link and will show as expired in the dashboard.' : '.'));
    }
  }

  const json = JSON.stringify(out, null, 1);
  // copy() is a DevTools convenience global, injected only into a console
  // opened directly in that panel — it throws ReferenceError anywhere else
  // (a docked/undocked console reattached differently, remote debugging,
  // some extensions' console proxies). Clipboard API needs a secure context
  // AND focus, neither guaranteed right after pasting into DevTools, so this
  // tries three ways and always leaves the JSON visible either way.
  let copied = false;
  try { copy(json); copied = true; } catch (e) { /* not in a real DevTools console */ }
  if (!copied) {
    try { await navigator.clipboard.writeText(json); copied = true; } catch (e) { /* no clipboard permission/focus */ }
  }
  console.log(`${copied ? 'Copied' : 'Could not auto-copy — select the JSON below and copy it manually:'} `
    + `${out.albums.length} albums and ${out.people.length} faces.`
    + (onAlbums ? ' Now run this again on /people to spot people with no album.' : ''));
  if (!copied) console.log(json);
})();

Photo sync

Health data (Health Connect)

A browser page has no way to read Health Connect directly — it's a native Android API with no web equivalent. Instead, a bridge app on your phone (e.g. HC Webhook / "Health Connect Webhook" from Google Play) posts whatever data you grant it to health.php on this same server. This panel is a raw viewer for now, showing exactly what's arrived, rather than a finished chart — the actual display gets built once a real payload's shape is known.
Setup: copy server/health.php.example to server/health.php, set $SECRET to the same value as sync.php, upload next to it. In the bridge app, point its webhook target at that URL — append ?secret=<your secret> to it if the app doesn't let you set a custom header.
